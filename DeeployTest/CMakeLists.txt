include_directories(${GENERATED_SOURCE})

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# Untiled baseline (see Platforms/GAP9/CMakeLists.txt). The define must be in
# scope for the *generated* network code and for deeploylib, not just for the
# test executable: the L1 allocations and the mchan calls it switches live
# there. Directory scope here covers every target created below.
if(DEEPLOY_L1_AS_L2)
  add_compile_definitions(DEEPLOY_L1_AS_L2)
  if(TARGET deeploylib)
    target_compile_definitions(deeploylib INTERFACE DEEPLOY_L1_AS_L2)
  endif()
endif()

if(TRAINING OR MEZO_TRAINING)  # QW: build training/optimizer OBJECT libs for both BP and MeZO (ZO) -- QW
  add_library(training_network OBJECT ${GENERATED_SOURCE}/TrainingNetwork.c)
  target_link_libraries(training_network PUBLIC deeploylib)
  # The codegen emits a #ifdef POWER_MEASUREMENT GPIO notch at the fwd/bwd
  # boundary inside RunTrainingNetwork; the training_network object must see the
  # define for it to compile in (the executable target alone is not enough).
  if(POWER_MEASUREMENT)
    target_compile_definitions(training_network PRIVATE POWER_MEASUREMENT)
  endif()
  # Optimizer network (SGD kernel, compiled separately to allow different prefix)
  add_library(optimizer_network OBJECT ${GENERATED_SOURCE}/OptimizerNetwork.c)
  # -- QW exp13/14: strict fp32 for the generated training/optimizer graphs (inline Quant/Dequant/SCE) -- QW
  if(DEEPLOY_STRICT_FP32)
    target_compile_options(training_network PRIVATE -fno-fast-math -ffp-contract=off)
    target_compile_options(optimizer_network PRIVATE -fno-fast-math -ffp-contract=off)
    message(STATUS "[QW strict-fp32] TrainingNetwork.c / OptimizerNetwork.c: -fno-fast-math -ffp-contract=off")
  endif()
  target_link_libraries(optimizer_network PUBLIC deeploylib)
else()
  add_library(network OBJECT ${GENERATED_SOURCE}/Network.c)
  target_link_libraries(network PUBLIC deeploylib)
endif()

if(platform STREQUAL MemPool)
  add_subdirectory(Platforms/MemPool)

elseif(platform STREQUAL SoftHier)
  add_subdirectory(Platforms/SoftHier)

elseif(platform STREQUAL Generic)
  add_subdirectory(Platforms/Generic)

elseif(DEEPLOY_ARCH STREQUAL CMSIS)
  if(platform STREQUAL QEMU-ARM)
    add_subdirectory(Platforms/QEMU_ARM)
  endif()

elseif(DEEPLOY_ARCH STREQUAL PULP)
  file(GLOB_RECURSE HEXLIST
    "${GENERATED_SOURCE}/hex/**"
  )

  if (NOT HEXLIST)
    if(TRAINING OR MEZO_TRAINING)  # QW: NOFLASH for both BP and MeZO training libs -- QW
      target_compile_options(training_network PUBLIC -DNOFLASH)
      target_compile_options(optimizer_network PUBLIC -DNOFLASH)
    else()
      target_compile_options(network PUBLIC -DNOFLASH)
    endif()
  else()
    gvsoc_flags_add_files_to_hyperflash(GVSOC_HEX_HYPERFLASH_FLAGS HEXLIST)
    list(APPEND GVSOC_EXTRA_FLAGS ${GVSOC_HEX_HYPERFLASH_FLAGS})
  endif()

  # SCHEREMO: Waive warnings
  # Pointer sign warnings are caused by the data width abstraction used in Deeploy. Signedness is not explicitly modelled, as this is handled by kernels
  if(TRAINING OR MEZO_TRAINING)  # QW: waive pointer-sign for both BP and MeZO training libs -- QW
    target_compile_options(training_network PRIVATE -Wno-pointer-sign)
    target_compile_options(optimizer_network PRIVATE -Wno-pointer-sign)
  else()
    target_compile_options(network PRIVATE -Wno-pointer-sign)
  endif()

  if(platform STREQUAL Siracusa OR platform STREQUAL Siracusa_w_neureka)
    add_subdirectory(Platforms/Siracusa)
  elseif(platform STREQUAL PULPOpen)
    add_subdirectory(Platforms/PULPOpen)
  endif()
elseif(DEEPLOY_ARCH STREQUAL SNITCH)
  add_subdirectory(Platforms/Snitch)
elseif(DEEPLOY_ARCH STREQUAL CHIMERA)
  add_subdirectory(Platforms/Chimera)
elseif(platform STREQUAL GAP9 OR platform STREQUAL GAP9_w_NE16)  # -- QW

  # Search for hex files generated by Python code generator
  # These files indicate L3 mode (external memory with readfs)
  file(GLOB_RECURSE HEXLIST
    "${GENERATED_SOURCE}/hex/*"
  )

  if (NOT HEXLIST)
    # L2 mode: No flash/readfs files
    # Data lives in L2 memory only
    if(TRAINING OR MEZO_TRAINING)  # QW: NOFLASH for both BP and MeZO (ZO) training libs on GAP9 -- QW
      target_compile_options(training_network PUBLIC -DNOFLASH)
      target_compile_options(optimizer_network PUBLIC -DNOFLASH)
    else()
      target_compile_options(network PUBLIC -DNOFLASH)
    endif()
    message(STATUS "[Deeploy GAP9] L2 mode: No hex files found, -DNOFLASH set")
    message(STATUS "[Deeploy GAP9] If you expect L3 mode, ensure Python codegen created hex files in ${GENERATED_SOURCE}/hex/")
  else()
    # L3 mode: Use flash with readfs files
    # Data will be loaded from external flash via readfs
    list(LENGTH HEXLIST HEXCOUNT)
    message(STATUS "[Deeploy GAP9] L3 mode: Found ${HEXCOUNT} hex file(s)")
    message(STATUS "==== HEXLIST ====")
    foreach(f ${HEXLIST})
      message(STATUS "  ${f}")
    endforeach()
    gvsoc_flags_add_files_to_flash(GAPY_RUNNER_ARGS HEXLIST)
    message(STATUS "GAPY_RUNNER_ARGS: ${GAPY_RUNNER_ARGS}")
  endif()

  # Waive warnings in generated code
  # RW: --profileTiling instrumentation emits machine-generated patterns that trip
  # several GCC -Werror classes (maybe-uninitialized on "..._measurements$" timing
  # vars, old-style-declaration from "static" not leading, ...). These are
  # generated, not hand-written, so demote ALL warnings to non-fatal for the
  # network objects via -Wno-error (warnings still print; no effect on the SDK or
  # on non-profiling builds, which simply produce no such warnings).
  if(TRAINING OR MEZO_TRAINING)  # QW: waive for both BP and MeZO (ZO) generated libs on GAP9 -- QW
    target_compile_options(training_network PRIVATE -Wno-pointer-sign -Wno-sign-compare -Wno-error)
    target_compile_options(optimizer_network PRIVATE -Wno-pointer-sign -Wno-sign-compare -Wno-error)
  else()
    target_compile_options(network PRIVATE -Wno-pointer-sign -Wno-sign-compare -Wno-error)
  endif()

  # RW 2026-10-13: the generated network code (inlined Quant/Dequant/RequantShift/SCE/transpose loops and
  # every closure) was still built at the SDK's -Os; with the slave stacks in L2 its spills cost L2 latency.
  # Same trick as TargetLibraries/GAP9: per-source -O3 is appended last and wins over -Os.
  option(GAP9_NETWORK_O3 "Compile the generated GAP9 network sources at -O3 (+~35 KB of L2 code; ResNet8 no longer fits L2)" OFF)
  if(GAP9_NETWORK_O3)
    if(TRAINING OR MEZO_TRAINING)
      set_source_files_properties(${GENERATED_SOURCE}/TrainingNetwork.c ${GENERATED_SOURCE}/OptimizerNetwork.c
                                  PROPERTIES COMPILE_OPTIONS "-O3")
    else()
      set_source_files_properties(${GENERATED_SOURCE}/Network.c PROPERTIES COMPILE_OPTIONS "-O3")
    endif()
  endif()

  add_subdirectory(Platforms/GAP9)
endif()
